from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Predefined FAQs in Hindi
questions = [
    "प्रवेश कैसे प्राप्त करें?",
    "प्रवेश के लिए कौन से दस्तावेज चाहिए?",
    "परीक्षा कब होगी?",
    "परीक्षा का समय सारणी कहाँ मिलेगी?",
    "फीस कितनी है?",
    "फीस कैसे जमा करें?",
    "छात्रावास की सुविधा उपलब्ध है?",
    "छात्रावास की फीस कितनी है?"
]

answers = [
    "प्रवेश के लिए विश्वविद्यालय की वेबसाइट पर आवेदन करें।",
    "प्रवेश के लिए अंकपत्र, आधार कार्ड और प्रमाण पत्र आवश्यक हैं।",
    "परीक्षा की तारीख विश्वविद्यालय की वेबसाइट पर घोषित की जाएगी।",
    "परीक्षा समय सारणी परीक्षा विभाग की वेबसाइट पर उपलब्ध है।",
    "फीस पाठ्यक्रम के अनुसार अलग-अलग होती है।",
    "फीस ऑनलाइन या विश्वविद्यालय कार्यालय में जमा कर सकते हैं।",
    "हाँ, विश्वविद्यालय में छात्रावास की सुविधा उपलब्ध है।",
    "छात्रावास की फीस विश्वविद्यालय के नियमों के अनुसार है।"
]

# TF-IDF vectorization
vectorizer = TfidfVectorizer()
question_vectors = vectorizer.fit_transform(questions)

# Chatbot
while True:
    user_query = input("\nअपना प्रश्न पूछें (बंद करने के लिए exit लिखें): ")

    if user_query.lower() == "exit":
        print("धन्यवाद!")
        break

    query_vector = vectorizer.transform([user_query])

    # Calculate cosine similarity
    similarity = cosine_similarity(query_vector, question_vectors)

    best_match = similarity.argmax()
    score = similarity[0][best_match]

    if score > 0:
        print("उत्तर:", answers[best_match])
    else:
        print("क्षमा करें, इस प्रश्न का उत्तर उपलब्ध नहीं है।")
